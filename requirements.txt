# ── 算法环境依赖（PC / 无板子阶段）─────────────────────────────

#   conda create -n rk python=3.10 -y && conda activate rk
#   pip install -r requirements.txt

# torch 走 CUDA 12.1 官方源（适配 RTX 2050 / Ampere）。
# 若机器无 NVIDIA 显卡，删掉下面这行 --extra-index-url，会自动装 CPU 版。
--extra-index-url https://download.pytorch.org/whl/cu121

torch==2.3.1

# 检测 / 姿态模型（自动下载 yolov8n.pt / yolov8n-pose.pt）
ultralytics==8.2.103

# 通用
opencv-python==4.10.0.84
numpy==1.26.4
matplotlib==3.9.2

# ── ONNX（为板端 RKNN 转换做准备）──────────────────────────────
# ⚠ numpy 必须钉在 1.26.x：torch 2.3.1 与 numpy 2.x 不兼容，装了 numpy 2 会报
#   `_ARRAY_API not found` → 运行时 "Numpy is not available"，ultralytics 直接崩。
#   而 pip 装 onnx 时会**顺手把 numpy 升到 2.x**（已实测踩过）。所以：
#       pip install onnx onnxruntime "numpy==1.26.4"
#   **三个一起装**，让 pip 在同一次解析里看到约束，numpy 就不会被升级。
#   （onnx 依赖的 ml-dtypes 会报 "requires numpy>=2.0.0" 的警告，实测无害，可忽略。）
onnx==1.23.0
onnxruntime==1.23.2

# rknn-toolkit2 仅 Linux（WSL2），T0.9 才用，这里不列，避免 Windows 安装失败。
